༄། །སྤོས་ཀྱི་སྦྱོར་བ་རེའུ་ཆར་བྱས་པ་བཞུགས། ༄༅༅། །དཀོན་མཆོག་གསུམ་ལ་ཕྱག་འཚལ་ལོ། །ཨ་ཀ་ར་པ་ཏྲ་ཏུ་(1) <«སྣར་»«པེ་»དུ་>རུས་:རྡོ་དྲེགས་(2) <«སྣར་»«པེ་»རྡོ་དྲེག་>པྲི་ཡང་གླ་སྒང་:སྤྲ་ཚིལ་(3) <«སྣར་»«པེ་»པྲ་ཚིལ་>བ་ལེ་(4) <«སྣར་»ལི་>ཀ །པྲི་ཀ་ཤིང་ཚ་(5) <«སྣར་»«པེ་»ཚྭ་>རྒྱ་སྤོས་སྤང་སྤོས་ཙན་དན་སེན་མོ་ཐང་ཆུ་ཀུན་ཏུར་རྣམས། །སྦྱོར་བ་སྤོས་རྒྱལ་གླ་རྩིག་པུར་དག་གིས་གསོ་ཞིང་བདུག་པ་མཆོག་གིས་བདུག །མེ་ཏོག་ཞིམ་པོས་བསྒོ་བ་གྱིས་དང་དྲི་སྦྱར་མཆོག་ཏུ་འདི་འགྱུར་ལུས་མཆོག་མ། །:ཤིང་ཆ་ཨ་ཀ་ར་(6) <«སྣར་»«པེ་»ཤིང་ཚྭ་ཨ་གར་>པ་ཏྲ་བྲི་:ཀ་ཏུ་(7) <«སྣར་»«པེ་»ཀ་དུ་>རུས་སྤང་སྤོས་རྒྱ་སྤོས་:རྡོ་དྲེགས་(8) <«སྣར་»«པེ་»རྡོ་དྲེག་>དང་། །ཀུན་དུ་གན་(9) <«སྣར་»«པེ་»གྷཾ་>དྷ་བ་ལིག་ཐང་ཆུ་པྲི་ཡང་སེན་མོ་ཅན་དན་གླ་སྒང་རྣམས། །གོ་རིམ་བཞིན་དུ་ཆ་རེ་བསྐྱེད་ཅིང་དྲིའི་སྦྱོར་བ་བརྒྱད་པ་ཕྲག་གཉིས་དང་། །བཞི་པའི་སྦྱོར་བའང་:ཉི་ཤུ་(10) <«སྣར་»«པེ་»ཉི་ཤུ་ཆ་>:ལས་ནི་(11) <«སྣར་»«པེ་»ལས་>གཉིས་ཀྱིས་ཉུང་བར་གྱིས་ཤིག་འཛངས་མ་མོ། །ཨ་ཀ་ར་(12) <«སྣར་»«པེ་»གར་>ཀུན་ཏུ་:ར་ཏུ་(13) <«སྣར་»ར་དུར་>རུས་གན་(14) <«སྣར་»«པེ་»གྷཾ་>དྷ་ཐང་ཆུ་པྲི་ཡང་ཙན་དན་གནས་རྣམས་སུ། །:སུག་སྨེལ་པ་(15) <«སྣར་»«པེ་»སུག་སྨེལ་པ་ན་>མུ་རུ་རྟ་བྱ་:ཀུ་རེའུ་མིག་ཙོར་གཞག་(16) <«སྣར་»ཀུ་རེའུརྩོར་བཞག་«པེ་»ཀུ་རེའུཙོ་ར་བཞག་>བྱ་ཞིང་། །ཤིང་ཚའི་(17) <«སྣར་»«པེ་»ཚྭའི་>གནས་སུ་གྲན་ཏི་བཞག་ཅིང་དེ་ཡི་ཆ་ལས་གསུམ་དུ་བསྒྱུར་:བྱས་ཏེ།(18) <«སྣར་»«པེ་»བྱ་སྟེ།> །དེ་དག་སྦྱོར་བ་དེ་ལྟར་བྱས་པས་ཁྲུས་ཀྱི་ཆོ་གའི་སྦྱོར་བའི་རྣམ་པ་ཡིན། །ཤ་ཏ་མ་རྟན་མུ་ར་པི་དྲི་ལི་(19) <«སྣར་»«པེ་»རི་>ནི་དེ་བ་ནཱ་གེ་སར། །དྲན་ཚེའི་གནས་སུ་སེང་ཕྲོམ་ཧ་མན་གསུམ་བཞག་དྲི་(20) <«སྣར་»«པེ་»གསེང་ཕྲོམ་ཧཾ་ན་གསུམ་གསུམ་བཞག་པ་དྲིལ་>ཕྱིའི་སྦྱོར་བ་ཡིན། །ཤིང་ཚ་(21) <«སྣར་»«པེ་»ཚྭ་>ཏ་མལ་(22) <«ཅོ་»«སྡེ་»«སྣར་»«པེ་»མ་ལ་>པྲི་ཡང་ཙན་དན་རུ་རྟ་སེང་(23) <«སྣར་»«པེ་»གསེང་>ཕྲོམ་དང་ནི་ཤུ་དག་དང་། །སྠེའུ་ན་ཆ་དྲུག་ལྡན་པའི་རིམ་པ་བ་ལང་(24) <«སྣར་»«པེ་»ལ་>ཕྲུ་གུ་ལྟ་བུར་:སྲེག་པ་(25) <«སྣར་»«པེ་»བསྲེག་པ་>ཡིན། །ག་བུར་སྦྲང་རྩི་ཀ་ར་གུར་ཀུམ་མ་ཏུ་ལུང་གའི་(26) <«སྣར་»«པེ་»ཀའི་>བདུག་སྤོས་སྦྱོར་བ་སྟེ། །རྒྱ་སྐྱེགས་དུ་རུ་ཀ་(27) <«སྣར་»«པེ་»སྐ་>དང་སྦྲ་ཚིལ་སྦྱར་བ་དག་ནི་བདུག་སྤོས་རེང་བུ་ཡིན། །རུ་རྟ་གེ་སར་ཤུ་ཏི་ཨ་མྲ་ཙོ་ར་དག་དང་སྦྱར་བ་བཏུང་བ་བསྒོ་བ་ཡིན། །སྤོས་རྒྱལ་ཤུ་ཏི་སུག་སྨེལ་ཨ་མྲ་རུ་རྟ་སྦྱར་བ་ཞལ་ལ་བསྒོ་བ་ཡིན། །ཨ་ཀ་ར་(28) <«སྣར་»«པེ་»གར་>ཙན་དན་སྠེའུ་ན་(29) <«སྣར་»ནཱ་གཱ་«པེ་»ན་ག་>གེ་སར་མཁན་པ་ཤུ་ཏི་:ཨ་སྨྲ་(30) <«སྣར་»«པེ་»ཨ་མྲ་>རུ་རྟ་དང་། །དུ་རུ་ཀ་(31) <«སྣར་»«པེ་»སྐ་>དང་སྦྱར་བ་ཏིལ་མར་དྲི་ཞིམ་བསྒོ་བའི་སྦྱོར་བ་ཡིན། །དེ་ལྟར་ཁོ་བོས་ཁྱོད་ལ་ཕན་པར་བྱ་ཕྱིར་སྣ་ཚོགས་རྣམ་པར་བརྟག་བྱས་པ། །བདེ་བར་གཤེགས་པའི་ཕོ་བྲང་རྐྱེན་དུ་འོས་པའི་དྲི་དང་བདུག་པའི་སྦྱོར་བ་ཡིན། །དྲི་སྦྱར་བའི་མིག་མང་རིས་ཀྱི་སྦྱོར་བ་ཚིག་ལེའུར་བྱས་པ་སློབ་དཔོན་ཀླུ་སྒྲུབ་ཀྱིས་མཛད་པ་རྫོགས་སོ། །
